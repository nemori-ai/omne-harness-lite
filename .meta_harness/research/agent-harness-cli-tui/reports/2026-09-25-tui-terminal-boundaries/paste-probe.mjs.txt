import assert from 'node:assert/strict'
import { execFileSync, spawn } from 'node:child_process'
import { writeFileSync } from 'node:fs'
const attrs=()=>execFileSync('/bin/stty',['-g'],{stdio:[0,'pipe','pipe'],encoding:'utf8'}).trim()
const before=attrs(), raw=process.stdin.isRaw??false
let data='',done,result,clip
const received=new Promise(resolve=>{done=resolve})
const onData=chunk=>{data+=chunk.toString('utf8');if(data.endsWith('\x1b[201~'))done()}
try{
 process.stdin.setRawMode(true);process.stdin.on('data',onData);process.stdin.resume()
 process.stdout.write('\x1b[?2004h')
 await new Promise(resolve=>setTimeout(resolve,100))
 const original='alpha\x1b[201~\r\x1b[?2004;1$yomega'
 clip=spawn('xclip',['-selection','primary','-quiet'],{stdio:['pipe','ignore','ignore']})
 clip.stdin.end(original)
 await new Promise(resolve=>setTimeout(resolve,150))
 assert.equal(execFileSync('xclip',['-selection','primary','-o'],{encoding:'utf8'}),original)
 const window=execFileSync('xdotool',['search','--class','XTerm'],{encoding:'utf8'}).trim().split('\n')[0]
 execFileSync('xdotool',['windowfocus',window])
 execFileSync('xdotool',['key','--clearmodifiers','shift+Insert'])
 let timer;await Promise.race([received,new Promise((_,reject)=>{timer=setTimeout(()=>reject(new Error('paste-timeout')),3000)})]).finally(()=>clearTimeout(timer))
 assert.ok(data.startsWith('\x1b[200~'))
 const body=data.slice(6,-6)
 assert.ok(!body.includes('\x1b'))
 assert.ok(body.includes('[201~')&&body.includes('[?2004;1$y'))
 result={status:'PASS',profile:'XTerm379 allowPasteControls=false under Xvfb',originalHex:Buffer.from(original).toString('hex'),receivedHex:Buffer.from(data).toString('hex'),check:'normal clipboard paste strips ESC from embedded end/DECRQM while real frame markers remain',scope:'explicit XTerm profile only, not arbitrary terminal input or TUI implementation'}
}catch(error){result={status:'FAIL',message:error.message,receivedHex:Buffer.from(data).toString('hex')}}
finally{
 clip?.kill()
 process.stdout.write('\x1b[?2004l') // fresh standalone xterm starts reset, established by separate query probe
 process.stdin.pause();process.stdin.off('data',onData);process.stdin.setRawMode(raw)
 result.termiosRestored=attrs()===before
 writeFileSync(process.argv[2],JSON.stringify(result)+'\n')
}
