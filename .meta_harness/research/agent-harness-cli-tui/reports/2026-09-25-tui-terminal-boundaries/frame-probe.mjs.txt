import assert from 'node:assert/strict'
import { Input, CURSOR_MARKER, visibleWidth, stripTerminalSequences, truncateToWidth } from '@earendil-works/pi-tui'
import xterm from '@xterm/headless'
const { Terminal } = xterm
const input = new Input({prompt: '> '}); input.focused=true
const identity=['Center: '+ 'c'.repeat(32),'Account: '+ 'a'.repeat(32),'Organization: '+ 'o'.repeat(32)]
function frame(columns,rows){
 const field=input.render(columns-1)[0], i=field.indexOf(CURSOR_MARKER)
 assert.ok(i>=0)
 const cursor=visibleWidth(field.slice(0,i))+1
 const safe=stripTerminalSequences(field.replace(CURSOR_MARKER,''))
 const lines=[...identity, 'Organization name:',safe,'Enter: preview | Esc: cancel']
 assert.ok(identity.every(line=>visibleWidth(line)<columns))
 assert.ok(lines.length<rows)
 return '\x1b[H\x1b[2J'+lines.join('\r\n')+`\x1b[5;${cursor}H`
}
for(const [cols,rows] of [[80,24],[60,16],[120,30]]) {
 const term=new Terminal({cols,rows,allowProposedApi:true})
 input.setValue('中文abcdef'.repeat(600))
 const bytes=frame(cols,rows)
 assert.ok(!/\x1b\[\?[0-9]+[hl]/.test(bytes))
 await new Promise(resolve=>term.write(bytes,resolve))
 for(let i=0;i<identity.length;i++) assert.equal(term.buffer.active.getLine(i).translateToString(true),identity[i])
 assert.equal(term.buffer.active.baseY,0)
 assert.equal(term.buffer.active.cursorY,4)
 assert.equal(input.getValue(),'中文abcdef'.repeat(600))
 term.dispose()
}
console.log(JSON.stringify({status:'PASS',piTui:'0.86.1',renderer:'application composes public Input.render + width utilities, no TUI/ProcessTerminal',sizes:[[80,24],[60,16],[120,30]],checks:['horizontal input does not scroll identity out of view','cursor follows focused component marker','no terminal private modes written','original input preserved'],scope:'headless emulator and public component probe only; not adopted terminal lifecycle, paste framing, actual PTY or IME'}))
