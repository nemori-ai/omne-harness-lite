import assert from 'node:assert/strict'
import { execFileSync } from 'node:child_process'
import { writeFileSync } from 'node:fs'
const out=process.argv[2], stage=process.argv[3]
const attrs=()=>execFileSync('/bin/stty',['-g'],{stdio:[0,'pipe','pipe'],encoding:'utf8'}).trim()
const before=attrs();let original,modeWait,buffer='',paste='',ime='',phase='query',result,changed=false
const raw=process.stdin.isRaw??false
const received=[]
let pasted,composed
const pasteDone=new Promise(r=>pasted=r),imeDone=new Promise(r=>composed=r)
function onData(bytes){
 received.push(bytes.toString('hex'))
 if(phase==='query'){
  buffer+=bytes.toString('latin1');const m=/\x1b\[\?2004;([0-4])\$y/.exec(buffer)
  if(m&&modeWait){const yes=modeWait;modeWait=undefined;buffer='';yes(Number(m[1]))}
 }else if(phase==='paste'){
  paste+=bytes.toString('utf8');if(paste.endsWith('\x1b[201~'))pasted()
 }else{ime+=bytes.toString('utf8');if(ime.includes('你好'))composed()}
}
async function bounded(promise,label){let timer;try{return await Promise.race([promise,new Promise((_,reject)=>timer=setTimeout(()=>reject(Error(label+' timeout')),15000))])}finally{clearTimeout(timer)}}
async function query(){phase='query';buffer='';return bounded(new Promise(r=>{modeWait=r;process.stdout.write('\x1b[?2004$p')}),'query')}
try{
 process.stdin.setRawMode(true);process.stdin.on('data',onData);process.stdin.resume()
 original=await query();assert.ok(original===1||original===2)
 if(original===2){process.stdout.write('\x1b[?2004h');changed=true}
 assert.equal(await query(),1)
 phase='paste';writeFileSync(stage,'paste-ready');process.stdout.write('\r\nOMNE own-window paste probe ready\r\n')
 await bounded(pasteDone,'paste');assert.ok(paste.startsWith('\x1b[200~'))
 const body=paste.slice(6,-6);assert.ok(!body.includes('\x1b'));assert.ok(body.includes('[201~X')&&body.includes('[?2004;1$yomega'))
 phase='ime';writeFileSync(stage,'ime-ready');process.stdout.write('\r\nOMNE own-window IME probe ready\r\n')
 await bounded(imeDone,'IME')
 result={status:'PASS',platform:process.platform,node:process.version,paste,ime,received,scope:'actual iTerm GUI ordinary Cmd-V and WeType Pinyin composition committed to raw PTY; source probe, not adopted TUI profile or full display acceptance'}
}catch(e){result={status:'FAIL',error:e.message,paste,ime,received}}
finally{
 try{if(changed){process.stdout.write('\x1b[?2004l');assert.equal(await query(),original)}}catch(e){result={...result,status:'FAIL',restoreError:e.message}}
 process.stdin.pause();process.stdin.off('data',onData);process.stdin.setRawMode(raw)
 result.termiosRestored=attrs()===before
 writeFileSync(out,JSON.stringify(result,null,2)+'\n');writeFileSync(stage,'done')
 process.stdout.write('\r\nOMNE source probe complete\r\n')
}
