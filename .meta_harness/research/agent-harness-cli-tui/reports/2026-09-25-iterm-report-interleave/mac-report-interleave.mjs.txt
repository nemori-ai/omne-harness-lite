import { writeFileSync } from 'node:fs'
import { execFileSync } from 'node:child_process'
import { InputFramer } from '/private/tmp/omne-cli-d053-cb_mt37l/app/dist/tui/framer.js'
const [out,stage]=process.argv.slice(2)
const before=execFileSync('/bin/stty',['-g'],{stdio:[0,'pipe','pipe'],encoding:'utf8'}).trim()
const raw=process.stdin.isRaw??false
let phase='query',reply,original,changed=false,sent=false,all=Buffer.alloc(0),records=[],events=[],result
const f=new InputFramer(e=>events.push(e))
let finished
const finish=new Promise(r=>finished=r)
function input(b){
 if(phase==='query') { const m=/\x1b\[\?2004;([0-4])\$y/.exec(b.toString('latin1'));if(m&&reply){const r=reply;reply=undefined;r(Number(m[1]))};return }

}catch(e){result={status:'FAIL',error:e.message,records,events,inPaste:f.inPaste}}
finally{

 try{if(changed){process.stdout.write('\x1b[?2004l');if(await query()!==original)throw Error('mode mismatch')}}catch(e){result.restoreError=e.message}
