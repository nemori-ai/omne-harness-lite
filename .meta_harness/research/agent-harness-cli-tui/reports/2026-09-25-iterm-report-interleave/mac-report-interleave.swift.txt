func apple(_ script: String) throws -> String {
 let p=Process();p.executableURL=URL(fileURLWithPath:"/usr/bin/osascript");p.arguments=["-e",script]
 let out=Pipe();p.standardOutput=out;try p.run();p.waitUntilExit()
 guard p.terminationStatus==0 else { throw NSError(domain:"AppleScript",code:Int(p.terminationStatus)) }
 return String(data:out.fileHandleForReading.readDataToEndOfFile(),encoding:.utf8)!.trimmingCharacters(in:.whitespacesAndNewlines)
}
func waitStage(_ wanted:String) throws {
 let end=Date().addingTimeInterval(20)
 while Date()<end {
  let s=(try? String(contentsOfFile:base+"/mac-interleave-r1.stage",encoding:.utf8)) ?? ""
  if s==wanted { return }; if s=="done" { throw NSError(domain:"probe-ended",code:1) }
  Thread.sleep(forTimeInterval:0.05)
 }
 throw NSError(domain:"stage-timeout",code:1)
}
func sourceID(_ source:TISInputSource)->String {
 let ptr=TISGetInputSourceProperty(source,kTISPropertyInputSourceID)!
 return Unmanaged<CFString>.fromOpaque(ptr).takeUnretainedValue() as String
}
let command=node+" "+base+"/mac-report-interleave.mjs "+base+"/mac-interleave-r1.json "+base+"/mac-interleave-r1.stage"

try window.write(toFile:base+"/mac-interleave-r1.window",atomically:true,encoding:.utf8)
defer { _ = try? apple("tell application \"iTerm\" to close window id "+window) }

// Preserve all pasteboard data only in memory; never serialize or log user clipboard contents.
let board=NSPasteboard.general
let saved=(board.pasteboardItems ?? []).map { item in item.types.compactMap { type -> (NSPasteboard.PasteboardType,Data)? in guard let data=item.data(forType:type) else{return nil};return(type,data) } }
board.clearContents();board.setString(String(repeating: "a", count: 758),forType:.string)
let ours=board.changeCount
func restoreClipboard() {

 board.clearContents();let items=saved.map { row -> NSPasteboardItem in let item=NSPasteboardItem();for(type,data)in row{item.setData(data,forType:type)};return item };board.writeObjects(items)
}
defer { restoreClipboard() }

guard result["status"] as? String == "OBSERVED" else {throw NSError(domain:"Source-probe-failed",code:1)}
print("Completed own iTerm window " + window + "; clipboard restored if unchanged; input source restored on exit")
