# ---- probe.py ----
import sys, tempfile, pathlib
sys.path.insert(0, "skills_ref_pkg")
from skills_ref.validator import validate
from skills_ref.parser import parse_frontmatter
cases = {
 "1 minimal": ("demo", "---\nname: demo\ndescription: d\n---\nbody\n"),
 "2 unknown top-level key": ("demo", "---\nname: demo\ndescription: d\nuser-invocable: true\n---\nbody\n"),
 "3 nested metadata value": ("demo", "---\nname: demo\ndescription: d\nmetadata:\n  deps:\n    tools: [a, b]\n---\nbody\n"),
 "4 list metadata value": ("demo", "---\nname: demo\ndescription: d\nmetadata:\n  tags: [a, b]\n---\nbody\n"),
 "5 unicode name": ("日本語", "---\nname: 日本語\ndescription: d\n---\nbody\n"),
 "6 name != dir": ("other", "---\nname: demo\ndescription: d\n---\nbody\n"),
 "7 consecutive hyphens": ("a--b", "---\nname: a--b\ndescription: d\n---\nbody\n"),
 "8 --- inside description": ("demo", "---\nname: demo\ndescription: see a---b here\n---\nbody\n"),
 "9 allowed-tools list": ("demo", "---\nname: demo\ndescription: d\nallowed-tools: [Read, Grep]\n---\nbody\n"),
 "10 empty body": ("demo", "---\nname: demo\ndescription: d\n---\n"),
 "11 unquoted colon": ("demo", "---\nname: demo\ndescription: Use when: pdf\n---\nbody\n"),
}
for label,(d,content) in cases.items():
    with tempfile.TemporaryDirectory() as t:
        p = pathlib.Path(t)/d; p.mkdir()
        (p/"SKILL.md").write_text(content)
        errs = validate(p)
        try:
            md, body = parse_frontmatter(content)
            extra = f"parsed={md!r} body={body!r}"
        except Exception as e:
            extra = f"parse_exc={e!r}"
        print(f"[{label}] errors={errs}\n    {extra}")
