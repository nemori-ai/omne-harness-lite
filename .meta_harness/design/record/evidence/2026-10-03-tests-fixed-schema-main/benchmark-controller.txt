"""Five paired measurements. Only baseline test files and cases are measured.

New #65 tests run separately as acceptance evidence. No tracked test is modified.
Every executed test identity and status is retained to audit equal workloads.
"""
import hashlib
import json
import os
from pathlib import Path
import re
import statistics
import subprocess
import sys
import time

root = Path(__file__).resolve().parents[3]
task = root / 'tmp/test-tiers-mutation/final65-main-integration'
copies = json.loads((task / 'clones-r1/result.json').read_text())['copies']
phase = sys.argv[1]
assert phase == 'full'
out = task / 'performance-r4' / phase
out.mkdir(parents=True, exist_ok=True)
assert not (out / 'result.json').exists(), 'Existing measurements must not be overwritten'
members = [x['path'] for x in json.loads((task / 'manual-plan.json').read_text())['plan']['build']]
baseline_files = subprocess.check_output(
    ['git', 'ls-tree', '-r', '--name-only', copies['baseline']['sha']], cwd=root, text=True,
).splitlines()
new_cases = [
    'D-116 native loading is confined to declared fixed-schema and build sites',
    'checks the D-116 resource leaf and permits only its finite generated entry',
]
case_pattern = '^(?![\\s\\S]*(?:' + '|'.join(re.escape(x) for x in new_cases) + '))'

def normalize_title(file, title):
    # This test prints a fresh process token inside an otherwise fixed fixture.
    if (file == '{ROOT}/packages/session/session-log-jsonl/tests/plugin.spec.ts'
            and title.startswith('actual Cordis 4.0.4 thin provider ownership invalid config fails before providing or acquiring files: ')):
        return re.sub(r'("current":"vitest-life:)[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}("})',
                      r'\1{PROCESS_UUID}\2', title)
    return title

workload = {}
for member in members:
    manifest = json.loads((Path(copies['baseline']['path']) / member / 'package.json').read_text())
    assert manifest['scripts']['test'] == 'vitest run', member
    files = [p[len(member) + 1:] for p in baseline_files if p.startswith(member + '/tests/')
             and re.search(r'\.(spec|test)(-d)?\.[cm]?[jt]sx?$', p)]
    assert files, member
    workload[member] = files
(out / 'workload.json').write_text(json.dumps({
    'files': workload, 'test_name_pattern': case_pattern,
    'reason': 'Use original tracked test files. The two new cases within existing tools test files are excluded only from performance measurements.',
}, indent=2) + '\n')
for label, copy in copies.items():
    path = Path(copy['path'])
    assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=path, text=True).strip() == copy['sha']
    assert subprocess.check_output(['git', 'status', '--porcelain'], cwd=path, text=True) == ''
    for member, files in workload.items():
        for file in files:
            assert (path / member / file).is_file(), label + ':' + member + '/' + file

result = {'phase': phase, 'copies': copies, 'cpu': 7, 'nice': 5,
          'member_concurrency': 1, 'workers': 1, 'samples': [],
          'scope': 'Prebuilt outputs. Fresh Node/Vitest process per sample/member. No build or machine queue in measured wall time.'}
reference_cases = None


def save():
    (out / 'result.json').write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n')


def execute(args, cwd, env, log):
    assert os.getpriority(os.PRIO_PROCESS, 0) == 0, 'The controller must not add priority increments'
    command = ['taskset', '-c', '7', 'nice', '-n', '5'] + args
    start = time.perf_counter()
    with log.open('wb') as stream:
        code = subprocess.run(command, cwd=cwd, env=env, stdout=stream, stderr=subprocess.STDOUT).returncode
    return {'command': command, 'cwd': str(cwd), 'exit_code': code,
            'wall_seconds': time.perf_counter() - start, 'log': str(log)}


def run_sample(label, pair):
    global reference_cases
    copy = copies[label]
    path = Path(copy['path'])
    folder = out / ('pair%d-%s' % (pair, label))
    folder.mkdir(exist_ok=False)
    env = dict(os.environ, VITEST_MAX_WORKERS='1', OMNE_TEST_MEMBER_CONCURRENCY='1',
               npm_config_workspace_concurrency='1', OMNE_TEST_TIER='fast',
               TMPDIR=copy['temp'], TMP=copy['temp'], TEMP=copy['temp'])
    for key in ['OMNE_TEST_SLOT_HELD', 'FULL']:
        env.pop(key, None)
    sample = {'label': label, 'pair': pair, 'steps': [], 'started_ns': time.time_ns()}
    result['samples'].append(sample)
    start = time.perf_counter()
    cases = []
    if phase in ['mrr', 'help']:
        args = (['node', '--input-type=module', '-e',
                 "await import('./packages/model/model-request-runtime/dist/index.js')"]
                if phase == 'mrr' else ['node', 'apps/cli/dist/main.js', '--help'])
        row = execute(args, path, env, folder / 'output.log')
        sample['steps'].append(row)
        sample['exit_code'] = row['exit_code']
        sample['wall_seconds'] = row['wall_seconds']
    else:
        targets = ['packages/model/pi-ai-transport'] if phase == 'pi' else members
        for member in targets:
            slug = member.replace('/', '__')
            report = folder / (slug + '.json')
            args = ['node', 'node_modules/vitest/vitest.mjs', 'run'] + workload[member]
            args += ['--testNamePattern', case_pattern, '--reporter=default', '--reporter=json',
                     '--outputFile.json=' + str(report)]
            row = execute(args, path / member, env, folder / (slug + '.log'))
            row['member'] = member
            sample['steps'].append(row)
            if report.exists():
                data = json.loads(report.read_text())
                for file in data.get('testResults', []):
                    for assertion in file.get('assertionResults', []):
                        title = normalize_title(file['name'].replace(str(path), '{ROOT}'), assertion['fullName'])
                        if any(new in title for new in new_cases):
                            assert assertion['status'] in ['pending', 'skipped', 'todo']
                            continue
                        cases.append([file['name'].replace(str(path), '{ROOT}'),
                                      title.replace(str(path), '{ROOT}').replace(copy['temp'], '{TMP}'),
                                      assertion['status']])
            save()
            print(json.dumps({'phase': phase, 'pair': pair, 'label': label,
                              'member': member, 'exit_code': row['exit_code'],
                              'wall_seconds': round(row['wall_seconds'], 3)}), flush=True)
            if row['exit_code']:
                break
        sample['wall_seconds'] = time.perf_counter() - start
        sample['exit_code'] = max(row['exit_code'] for row in sample['steps'])
        cases.sort()
        (folder / 'cases.json').write_text(json.dumps(cases, ensure_ascii=False, indent=1) + '\n')
        sample['case_count'] = len(cases)
        sample['passed'] = sum(c[2] == 'passed' for c in cases)
        sample['case_sha256'] = hashlib.sha256(json.dumps(cases, ensure_ascii=False).encode()).hexdigest()
        if reference_cases is None and not sample['exit_code']:
            reference_cases = cases
        sample['same_cases_and_statuses'] = cases == reference_cases
        if not sample['same_cases_and_statuses']:
            sample['workload_error'] = 'Test identities/statuses differ from first valid baseline sample'
            sample['exit_code'] = sample['exit_code'] or 2
    save()
    print(json.dumps({k: v for k, v in sample.items() if k != 'steps'}), flush=True)
    if sample['exit_code']:
        raise SystemExit(sample['exit_code'])



for pair in range(1, 6):
    order = ['baseline', 'candidate'] if pair % 2 else ['candidate', 'baseline']
    for label in order:
        run_sample(label, pair)
result['medians'] = {label: statistics.median(s['wall_seconds'] for s in result['samples']
                                             if s['label'] == label) for label in copies}
result['complete'] = True
save()
print(json.dumps({'phase': phase, 'medians': result['medians'], 'complete': True}), flush=True)
