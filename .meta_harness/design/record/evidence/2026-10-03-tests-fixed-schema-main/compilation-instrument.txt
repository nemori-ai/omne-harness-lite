// Measurement only. Preload the same installed classes and observe their methods,
// including calls from native ESM consumers that bypass Module._load hooks.
import { createRequire } from 'node:module'
import { resolve } from 'node:path'
import { performance } from 'node:perf_hooks'
import { writeFileSync } from 'node:fs'
const output = process.env.OMNE_65_COMPILE_PROFILE
if (!output) throw new Error('OMNE_65_COMPILE_PROFILE is required')
const require = createRequire(resolve('packages/core/identity-values/package.json'))
const seen = new WeakSet()
const record = { observedInstances: 0, compileCalls: 0, schemaCompilations: [] }
for (const specifier of ['ajv', 'ajv/dist/2020.js']) {
  const loaded = require(specifier)
  const Actual = loaded.default ?? loaded
  for (const method of ['compile', 'addSchema', 'getSchema', '_compileSchemaEnv']) {
    const original = Actual.prototype[method]
    if (typeof original !== 'function') throw new Error('Missing Ajv method: ' + method)
    Object.defineProperty(Actual.prototype, method, {
      configurable: true, writable: true,
      value: function (...args) {
        if (!seen.has(this)) {
          seen.add(this)
          record.observedInstances += 1
        }
        if (method === 'compile') record.compileCalls += 1
        const start = performance.now()
        try {
          return original.apply(this, args)
        } finally {
          if (method === '_compileSchemaEnv')
            record.schemaCompilations.push({
              id: args[0]?.schema?.$id ?? args[0]?.schema?.$ref ?? null,
              milliseconds: performance.now() - start,
              stack: new Error().stack.split('\n').slice(2, 8),
            })
        }
      },
    })
  }
}
process.on('exit', code => {
  writeFileSync(output, JSON.stringify({ ...record, exit_code: code }, null, 2) + '\n')
})
