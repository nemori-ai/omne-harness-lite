#!/usr/bin/env python3
"""Snapshot this task's actual inputs and all unique local evidence; no product writes."""
import hashlib
import json
import shutil
import subprocess
from pathlib import Path

repo = Path(__file__).resolve().parents[2]
destination = repo / "tmp/w1-public-consumers"
design = repo / ".meta_harness/design"
nodes = ["kernel-basis", "identity-values", "offer-expiry", "storage", "content-blocks"]

def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def git(*args):
    return subprocess.check_output(["git", *args], cwd=repo, text=True).strip()

catalog = json.loads((design / "registry/catalog.json").read_text())
frozen = json.loads((design / "registry/frozen.json").read_text())["files"]
contracts = [item for item in catalog["contracts"]
             if item["coordinate"]["contract_id"] in
             {"data.identity-values", "data.offer-expiry", "data.content-blocks"}]
assert len(contracts) == 3
inputs = []
for item in contracts:
    assert item["status"] == "accepted"
    assert item["coordinate"]["version"] == 1
    for resource in item["resources"]:
        sha = digest(design / resource)
        assert sha == frozen[resource], resource
        inputs.append({"path": ".meta_harness/design/" + resource, "sha256": sha,
                       "coordinate": item["coordinate"], "frozen": True})

paths = ["package.json", "pnpm-workspace.yaml", "pnpm-lock.yaml", "tsconfig.base.json",
         ".meta_harness/design/registry/catalog.json", ".meta_harness/design/registry/frozen.json"]
for node in nodes:
    paths.extend(f".meta_harness/design/realization/packages/core/{node}/{name}"
                 for name in ["README.md", "design.md", "verification.md", "bindings.yaml"])
for decision in ["cordis-adoption-version", "infrastructure-persistence-choices"]:
    paths.append(f".meta_harness/design/record/decisions/{decision}/decision.md")
paths.extend(item["path"] for item in inputs)
paths.extend(git("ls-files", "tools/core-verification").splitlines())
paths = sorted(set(paths))
for relative in paths:
    target = destination / "inputs" / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(repo / relative, target)

artifact = repo / "tools/core-verification/dist/w1-public-consumer.js"
artifact_snapshot = destination / "artifacts/w1-public-consumer.js"
artifact_snapshot.parent.mkdir(parents=True, exist_ok=True)
shutil.copyfile(artifact, artifact_snapshot)

products = []
for node in nodes:
    path = f"packages/core/{node}"
    products.append({"node": node, "lastPathCommit": git("log", "-1", "--format=%H", "--", path),
                     "trackedFiles": [{"path": relative, "sha256": digest(repo / relative)}
                                      for relative in git("ls-files", path).splitlines()],
                     "builtFiles": [{"path": str(file.relative_to(repo)), "sha256": digest(file)}
                                    for file in sorted((repo / path / "dist").rglob("*")) if file.is_file()]})

result = {
    "format": 1, "baseline": "198648965cb89100386cf8600451f7afa54e786a",
    "sourceRevision": git("rev-parse", "HEAD"), "branch": git("branch", "--show-current"),
    "trackedStatus": git("status", "--short"),
    "nodeVersion": subprocess.check_output(["node", "--version"], text=True).strip(),
    "pnpmVersion": subprocess.check_output(["pnpm", "--version"], text=True).strip(),
    "toolVersions": {name: json.loads((repo / "tools/core-verification/node_modules" / name / "package.json").read_text())["version"]
                     for name in ["typescript", "vitest", "@deepseek-ai/cordis", "kysely", "zod"]},
    "acceptedInputs": inputs,
    "configurationAndToolSources": [{"path": relative, "sha256": digest(repo / relative)} for relative in paths],
    "products": products,
    "changesSinceBaseline": git("diff", "--name-only", "19864896", "HEAD").splitlines(),
    "executionMetadata": [{"path": str(file.relative_to(repo)), "value": json.loads(file.read_text())}
                          for file in sorted(destination.glob("*/execution.json"))],
    "currentConsumerArtifact": {"path": "tools/core-verification/dist/w1-public-consumer.js", "sha256": digest(artifact),
                                "snapshot": str(artifact_snapshot.relative_to(repo)), "snapshotPurpose": "Exact executed bytes retained as evidence; rerun through the built member's original path so workspace package resolution is preserved."},
    "latestVerification": {"publicConsumerAndDependencies": {"log": "test-public-closure.log", "report": "test-results-public-closure.json", "parentRuntimeDependencyAndTypeTests": 10},
                           "retainedStorage": {"log": "test-final.log", "report": "test-results-final.json", "parentStorageSelftests": 2},
                           "note": "Executions preceded commits: initial execution.json records HEAD 19864896; dependency follow-up executions record HEAD d1404a93; latest public closure executions record HEAD f1b5e7e2 with task changes now persisted in the current revision. Current source snapshot and consumer artifact hash identify persisted bytes, not retroactively a clean commit execution."},
    "adoptedRealizationClarification": {"sourceCommit": git("rev-parse", "e6c9d66a"),
                                      "sourcePath": ".meta_harness/design/realization/packages/core/storage/design.md",
                                      "snapshot": "tmp/w1-public-consumers/storage-design-e6c9d66a.md",
                                      "sha256": digest(destination / "storage-design-e6c9d66a.md"),
                                      "adoption": "Root dispatcher confirmed D-026 approved scope and technically accepted mapping clarification. The root protocol consumes Kysely types; ./conformance consumes runtime sql and Vitest. This supersedes the too-broad dependency sentence in the baseline realization snapshot only, without a contract or dependency-exception expansion."},
    "preservedDependencyFailure": {"log": "test-dependencies.log", "report": "test-results-dependencies.json", "passed": 9, "failed": 1,
                                  "cause": "Baseline mapping said only Kysely types for the entire node; approved conformance runtime graph contains Kysely sql. Expected subpath edge was corrected only after dispatcher adoption and e6c9d66a mapping clarification. Root graph expectation stayed unchanged."},
    "notExecuted": ["full make check", "formal tool reruns", "PostgreSQL", "deployment cross-process protocols", "external JSON hops"],
    "firstMeasurementFailure": {"log": "typecheck-first.log", "cause": "report record type narrowing in new tool test", "initialCompleteSourceSnapshotPreserved": False},
}
(destination / "metadata.json").write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n")
unique = [file for file in sorted(destination.rglob("*")) if file.is_file() and file.name != "SHA256SUMS.json"]
manifest = [{"path": str(file.relative_to(repo)), "sha256": digest(file)} for file in unique]
# Existing storage self-tests save raw files elsewhere; explicitly retain them too.
for file in sorted((repo / "tmp/core-verification").rglob("*")):
    if file.is_file():
        manifest.append({"path": str(file.relative_to(repo)), "sha256": digest(file)})
(destination / "SHA256SUMS.json").write_text(json.dumps(manifest, ensure_ascii=False, indent=2) + "\n")
for item in manifest:
    assert digest(repo / item["path"]) == item["sha256"]
print(json.dumps({"sourceRevision": result["sourceRevision"], "acceptedFrozenInputs": len(inputs), "uniqueEvidenceFiles": len(manifest)}, ensure_ascii=False))
