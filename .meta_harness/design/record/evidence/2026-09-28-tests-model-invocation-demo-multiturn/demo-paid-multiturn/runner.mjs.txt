import assert from 'node:assert/strict'
import { randomUUID } from 'node:crypto'
import { spawn } from 'node:child_process'
import { readFile, readdir } from 'node:fs/promises'
import { join, resolve } from 'node:path'
import { openProductionLedger, privateOutputDirectory, writePrivateEvidence } from '../../tools/model-probe/src/production-gate.mjs'
import { estimateReservation } from '../../tools/model-probe/src/budget.mjs'
import { getPublic, readCredential } from '../../tools/model-probe/src/network.mjs'
import { plainUsd } from './m1-real/artifact/money.mjs'

const taskRoot = resolve('tmp/model-request-runtime')
const keyFile = process.argv[2]
if (!keyFile) throw new Error('Explicit credential file required')
const root = await privateOutputDirectory(join(taskRoot, `demo-multiturn-${randomUUID()}`))
const data = join(root, 'data')
const model = 'deepseek/deepseek-v4-pro-0813', provider = 'deepinfra/fp8'
let secret, ledger, current
const turns = []
function clean(text) {
  if (secret && text.includes(secret)) throw new Error('Credential present in evidence')
  return text
}
async function evidence(name, value) {
  await writePrivateEvidence(join(root, name), clean(typeof value === 'string' ? value : JSON.stringify(value, null, 2) + '\n'))
}
async function replies() {
  const found = []
  async function walk(path) {
    for (const entry of await readdir(path, { withFileTypes: true })) {
      const child = join(path, entry.name)
      if (entry.isDirectory()) await walk(child)
      else if (entry.name === 'entries.jsonl') {
        for (const line of (await readFile(child, 'utf8')).split('\n')) if (line) {
          const batch = JSON.parse(line, (key, value, context) => key === 'cost_usd' && typeof value === 'number' ? context.source : value)
          if (batch.entries?.length !== 2) continue
          const [message, response] = batch.entries.map(entry => entry.payload?.value)
          if (message?.type === 'assistant' && response?.replyDisposition === 'persisted') found.push({ message, response, batch })
        }
      }
    }
  }
  await walk(join(data, 'session-logs'))
  return found
}
function launch(label) {
  const command = `node examples/model-invocation-minimal-demo/dist/cli.js --data '${data}'`
  const child = spawn('script', ['-qefc', command, '/dev/null'], { cwd: process.cwd(), env: { ...process.env, OPENROUTER_API_KEY: secret }, stdio: ['pipe', 'pipe', 'pipe'] })
  let output = '', error = '', closed = false, code, signal
  const chunks = [], watchers = new Set()
  const wake = () => { for (const fn of watchers) fn() }
  child.stdout.on('data', chunk => { output += chunk.toString(); chunks.push({ at: new Date().toISOString(), bytes: chunk.length }); if (output.length > 250000) child.kill('SIGINT'); wake() })
  child.stderr.on('data', chunk => { error += chunk.toString(); wake() })
  child.on('error', () => { error += 'child process error'; wake() })
  child.on('close', (c, s) => { closed = true; code = c; signal = s; wake() })
  async function wait(predicate) {
    if (predicate()) return
    await new Promise((done, reject) => {
      const check = () => {
        if (predicate()) { clearTimeout(timer); watchers.delete(check); done() }
        else if (closed) { clearTimeout(timer); watchers.delete(check); reject(new Error('Demo exited before expected prompt')) }
      }
      const timer = setTimeout(() => { watchers.delete(check); child.kill('SIGINT'); reject(new Error('Demo prompt timeout')) }, 180000)
      watchers.add(check); check()
    })
  }
  return {
    async ready(n) { await wait(() => (output.match(/You> /g) ?? []).length >= n) },
    send(text) { child.stdin.write(text + '\n') },
    snapshot() { return { label, output, error, chunks, closed, code, signal } },
    async close() { if (!closed) child.stdin.write('/exit\n'); await wait(() => closed); return this.snapshot() },
    stop() { if (!closed) child.kill('SIGINT') },
  }
}
try {
  const facts = await getPublic(`/api/v1/models/${model}/endpoints`)
  const endpoint = JSON.parse(facts.raw).data.endpoints.find(row => row.tag === provider)
  assert.ok(endpoint)
  assert.equal(endpoint.context_length, 1048576)
  assert.ok(endpoint.max_completion_tokens >= 1024)
  const estimateUsd = estimateReservation(endpoint.pricing, 1048576, 1024)
  await evidence('price.json', { observedAt: facts.observedAt, model, provider, pricing: endpoint.pricing, estimateUsd })
  const ledgerPath = join(taskRoot, 'ledger.json')
  ledger = await openProductionLedger(ledgerPath, ledgerPath, { limitUsd: '100', researchLimitUsd: '50' })
  secret = await readCredential(keyFile)
  const prompts = ['Remember this exact code for our conversation: ORCHID-742. Reply with that code only.', 'What exact code did I ask you to remember? Reply with that code only.', 'After restarting, what exact code did I ask you to remember? Reply with that code only.']
  current = launch('same-process-two-turns')
  await current.ready(1)
  for (let turn = 0; turn < prompts.length; turn++) {
    if (turn === 2) {
      const first = await current.close()
      await evidence('terminal-first.json', first)
      assert.equal(first.code, 0)
      current = launch('reopened-third-turn')
      await current.ready(1)
    }
    const reservation = await ledger.reserve({ model, provider, stage: 'acceptance', estimateUsd })
    let actualUsd
    try {
      current.send(prompts[turn])
      await current.ready(turn === 2 ? 2 : turn + 2)
      const rows = await replies()
      const reply = rows[turn]
      // Capture observable output and durable facts BEFORE settlement or assertions.
      await evidence(`turn-${turn + 1}-terminal.json`, current.snapshot())
      await evidence(`turn-${turn + 1}-durable.json`, rows)
      if (reply?.response.cost?.source === 'upstream_usage') actualUsd = plainUsd(reply.response.cost.cost_usd)
      await ledger.settle(reservation, actualUsd)
      assert.equal(rows.length, turn + 1)
      assert.equal(reply.response.outcome, 'complete')
      assert.equal(reply.response.route.provider, provider)
      assert.equal(reply.batch.entries.length, 2)
      const answer = reply.message.blocks.filter(block => block.kind === 'text').map(block => block.text).join('')
      assert.ok(answer.includes('ORCHID-742'))
      assert.ok(current.snapshot().output.includes('Cumulative'))
      turns.push({ turn: turn + 1, reservation, actualUsd, outcome: reply.response.outcome, usage: reply.response.usage, answer })
      await evidence(`turn-${turn + 1}-result.json`, turns.at(-1))
    } catch (error) {
      await evidence(`turn-${turn + 1}-failure.json`, { message: error.message, terminal: current.snapshot(), actualUsd })
      if (ledger.snapshot().entries.find(entry => entry.id === reservation)?.state === 'reserved') {
        try { await ledger.settle(reservation, actualUsd) } catch {}
      }
      throw error
    }
  }
  const final = await current.close()
  await evidence('terminal-reopened.json', final)
  assert.equal(final.code, 0)
  assert.equal((await replies()).length, 3)
  await evidence('ledger-after.json', ledger.snapshot())
  await evidence('result.json', { pass: true, model, provider, turns, scope: 'Two turns in one real PTY process; third turn after close/reopen; actual OpenRouter usage/cost and atomic durable reply/response pairs. Not full 13-operation or 37-model certification.' })
  process.stdout.write(JSON.stringify({ root, pass: true, turns: turns.map(({ turn, actualUsd }) => ({ turn, actualUsd })) }) + '\n')
} catch (error) {
  current?.stop()
  await evidence('failure.json', { message: error.message, turns })
  process.stderr.write(JSON.stringify({ root, pass: false, message: clean(error.message) }) + '\n')
  process.exitCode = 1
} finally {
  if (ledger) await ledger.close()
}
