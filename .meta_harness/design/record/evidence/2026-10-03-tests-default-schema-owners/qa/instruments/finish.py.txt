"""Criterion 6 public types and dispatched one-time original MRR slow combination.
Controller nice0; each actual check is CPU7/nice5 once, no product writes.
Preserves originals and never changes assertions or configured test timeouts.
"""
from pathlib import Path
import json,os,subprocess,time,hashlib,sys
QA=Path(__file__).resolve().parent.parent
config=json.loads((QA/'freeze.json').read_text())
if not config['execution_lane_transferred']:raise SystemExit('No testing lane')
root=Path(config['candidate_root']);baseline=Path(config['baseline_root'])
e=QA/'evidence'/'finish';e.mkdir(exist_ok=False)
env=dict(os.environ,TMPDIR=config['tmpdir'],TMP=config['tmpdir'],TEMP=config['tmpdir'],LANG='C.UTF-8',LC_ALL='C.UTF-8',VITEST_MAX_WORKERS='1',OMNE_TEST_MEMBER_CONCURRENCY='1',npm_config_workspace_concurrency='1')
if 'OMNE_TEST_SLOT_HELD' in env:raise SystemExit('Slot wrapper forbidden')
commands=[]
def run(name,args,cwd,extra=None,timeout=360):
    command=['taskset','-c','7','nice','-n','5']+args
    out=e/(name+'.log');started=time.monotonic()
    with out.open('wb') as stream:
        try:code=subprocess.run(command,cwd=cwd,env=dict(env,**(extra or {})),stdout=stream,stderr=subprocess.STDOUT,timeout=timeout).returncode
        except subprocess.TimeoutExpired:code=124
    record=dict(name=name,command=command,cwd=str(cwd),exit_code=code,wall_seconds=round(time.monotonic()-started,3),sha256=hashlib.sha256(out.read_bytes()).hexdigest(),raw=out.name)
    commands.append(record);(e/'commands.json').write_text(json.dumps(commands,indent=2)+'\n')
    print('PASS' if code==0 else 'FAIL',name,flush=True)
    return code
template=(QA/'scenarios/public-types.template.ts').read_text()
for name,checkout in [('candidate',root),('baseline',baseline)]:
    directory=e/('types-'+name);directory.mkdir()
    (directory/'public-types.mts').write_text(template.replace('@CANDIDATE@',str(checkout)))
    (directory/'tsconfig.json').write_text(json.dumps(dict(extends=str(checkout/'tsconfig.base.json'),compilerOptions=dict(noEmit=True,typeRoots=[str(checkout/'packages/core/asset-evaluation/node_modules/@types')]),include=['public-types.mts']),indent=2)+'\n')
    code=run('public-types-'+name,['node',str(checkout/'packages/core/asset-evaluation/node_modules/typescript/bin/tsc'),'-p',str(directory/'tsconfig.json')],checkout)
    if code:raise SystemExit('FAIL public types: stop for triage and preserve evidence')
member=root/'packages/model/model-request-runtime'
report=e/'mrr-original-slow.json'
code=run('mrr-original-slow',['node','node_modules/vitest/vitest.mjs','run','--reporter=default','--reporter=json','--outputFile.json='+str(report)],member,{'OMNE_TEST_TIER':'slow'},1200)
result=dict(status='PASS' if code==0 else 'FAIL',commit=config['candidate_commit'],baseline=config['baseline_commit'],attempts=1,commands=commands,original_timeout_unchanged=True)
if report.exists():
    parsed=json.loads(report.read_text())
    result['tests']={key:parsed.get(key) for key in ['numTotalTests','numPassedTests','numFailedTests','numPendingTests','success']}
    result['failed']=[dict(file=suite['name'],name=test['fullName'],messages=test.get('failureMessages',[])) for suite in parsed['testResults'] for test in suite['assertionResults'] if test['status']=='failed']
    result['t15']=[dict(file=suite['name'],name=test['fullName'],status=test['status']) for suite in parsed['testResults'] if suite['name'].endswith('acceptance-t15-fresh-dispatch-inputs.spec.ts') for test in suite['assertionResults']]
(e/'result.json').write_text(json.dumps(result,indent=2)+'\n')
print(result['status'],'required final checks',flush=True)
sys.exit(0 if code==0 else 1)
