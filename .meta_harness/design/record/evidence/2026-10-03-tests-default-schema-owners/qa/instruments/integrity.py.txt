"""Read-only final coordinate/artifact check; writes QA evidence only.
Run on CPU7/nice5 after all product processes have ended.
"""
from pathlib import Path
import subprocess,json,hashlib,os,datetime
QA=Path(__file__).resolve().parent.parent
config=json.loads((QA/'freeze.json').read_text())
root=Path(config['candidate_root']);baseline=Path(config['baseline_root'])
e=QA/'evidence';sha=lambda path:hashlib.sha256(path.read_bytes()).hexdigest()
states=[]
for name,checkout in [('candidate',root),('baseline',baseline)]:
    head=subprocess.check_output(['git','rev-parse','HEAD'],cwd=checkout,text=True).strip()
    status=subprocess.check_output(['git','status','--porcelain=v1','--untracked-files=all'],cwd=checkout,text=True)
    assert head==config[name+'_commit'] and not status,(head,status)
    tracked=subprocess.check_output(['git','ls-files','-z'],cwd=checkout).decode().split('\0')
    files={}
    for relative in tracked:
        if relative.startswith(('packages/','apps/','tools/','scripts/')) or relative in ['package.json','pnpm-lock.yaml','pnpm-workspace.yaml','tsconfig.base.json','vitest.tiers.mts']:
            path=checkout/relative
            if path.is_file():files[relative]=sha(path)
    states.append(dict(copy=name,commit=head,git_status=status,tracked_input_files=files))
builds=json.loads((root/'tmp/test-tiers-mutation/default-owners-final-builds.json').read_text())
artifact_checks=[]
for member in builds['members']:
    files=member['candidate_files']
    for relative,digest in files.items():assert sha(root/member['member']/relative)==digest
    if not member['rebuilt_child']:
        for relative,digest in files.items():assert sha(baseline/member['member']/relative)==digest
    artifact_checks.append(dict(member=member['member'],files=len(files),rebuilt=member['rebuilt_child'],matches_declared_snapshot=True))
old={}
for run in ['qa65-main-r1','qa65-main-r2']:
    for name in ['report.md','report.json']:
        path=root/'tmp/test-tiers-mutation'/run/'evidence'/name
        old[run+'/'+name]=sha(path)
assert old['qa65-main-r1/report.md']=='46a69ac67bb77ab7954eb264a24f83dbd0dcf2357ea00531b8977d72f435eb54'
assert old['qa65-main-r1/report.json']=='53ad8bcc6c9f4dd026e23f324fb2ba15268a3e1e986dda003180496b8c777514'
assert old['qa65-main-r2/report.md']=='16a4876b648c03d9d7615539cc69a9114af5e7d85309d807446ba31ddb155053'
assert old['qa65-main-r2/report.json']=='14879770dffb99e779349f6005d9b68ec8ece791a0c604f8f366a08a0633ed68'
evidence_sources=['default-owners-self-r3/result.json','default-owners-self-r4/result.json','default-owners-self-r5/result.json','default-owners-self-r6/result.json','default-owners-common-r1/result.json','default-owners-common-r1/loop/status.json','default-owners-final-builds.json','default-owners-structural-disposition.md','default-owners-clone-comparison.json','default-owners-test-account.json']
source_records={name:sha(root/'tmp/test-tiers-mutation'/name) for name in evidence_sources}
result=dict(status='PASS',states=states,artifact_checks=artifact_checks,prior_reports_sha256=old,preparation_evidence_sha256=source_records,cpu_affinity=sorted(os.sched_getaffinity(0)),nice=os.getpriority(os.PRIO_PROCESS,0),completed_utc=datetime.datetime.now(datetime.timezone.utc).isoformat())
(e/'integrity.json').write_text(json.dumps(result,indent=2)+'\n')
print('PASS final integrity',len(artifact_checks),'members',sum(x['files'] for x in artifact_checks),'artifacts')
