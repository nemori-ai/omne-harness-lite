"""Preparation for ROOT's six L2 criteria. Do not execute without frozen lane.
Only writes this QA directory; never builds, installs, or changes tracked inputs.
Fresh native process per case; caller invokes this controller without nice nesting.
"""
from pathlib import Path
import json,os,subprocess,sys,time,hashlib

SCENARIOS=Path(__file__).resolve().parent
QA=SCENARIOS.parent
configuration=json.loads((QA/'freeze.json').read_text())
if not configuration['execution_lane_transferred']:
    raise SystemExit('Preparation only: ROOT has not transferred a frozen execution lane')
for key in ['candidate_commit','baseline_commit','candidate_root','baseline_root','tmpdir']:
    if not configuration.get(key):raise SystemExit('Missing frozen facility input: '+key)
if os.environ.get('OMNE_TEST_SLOT_HELD'):
    raise SystemExit('Slot wrapper is forbidden')
for name in ['candidate','baseline']:
    root=Path(configuration[name+'_root'])
    head=subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip()
    status=subprocess.check_output(['git','status','--porcelain=v1','--untracked-files=all'],cwd=root,text=True)
    if head!=configuration[name+'_commit'] or status:raise SystemExit('Frozen checkout coordinate or cleanliness mismatch: '+name)
run_id=os.environ.get('QA65_RUN_ID','initial')
if not run_id.replace('-','').isalnum():raise SystemExit('Use a simple unique run id')
evidence=QA/'evidence'/run_id
evidence.mkdir(parents=True,exist_ok=False)
environment=dict(os.environ,QA65_EXECUTION_ALLOWED='frozen-lane-transferred',LANG='C.UTF-8',LC_ALL='C.UTF-8',
    VITEST_MAX_WORKERS='1',OMNE_TEST_MEMBER_CONCURRENCY='1',npm_config_workspace_concurrency='1',
    TMPDIR=configuration['tmpdir'],TMP=configuration['tmpdir'],TEMP=configuration['tmpdir'])
records=[]
def invoke(name,arguments,root,timeout=180):
    output=evidence/(name+'.log')
    command=['taskset','-c','7','nice','-n','5','node','--experimental-transform-types','--experimental-import-meta-resolve']+arguments
    started=time.monotonic()
    with output.open('wb') as stream:
        try:code=subprocess.run(command,cwd=root,env=environment,stdout=stream,stderr=subprocess.STDOUT,timeout=timeout).returncode
        except subprocess.TimeoutExpired:code=124
    records.append(dict(name=name,command=command,cwd=str(root),exit_code=code,wall_seconds=round(time.monotonic()-started,3),raw=output.name,sha256=hashlib.sha256(output.read_bytes()).hexdigest()))
    (evidence/'commands.json').write_text(json.dumps(records,indent=2)+'\n')
    print('PASS' if code==0 else 'FAIL',name,flush=True)
    return code==0

candidate=Path(configuration['candidate_root']);baseline=Path(configuration['baseline_root'])
database=evidence/'seed.sqlite';metadata=evidence/'seed.json'
seed_from=os.environ.get('QA65_SEED_FROM_RUN')
if seed_from:
    if not seed_from.replace('-','').isalnum():raise SystemExit('Invalid saved seed run id')
    database=QA/'evidence'/seed_from/'seed.sqlite';metadata=QA/'evidence'/seed_from/'seed.json'
    if not database.exists() or not metadata.exists():raise SystemExit('Missing preserved seed')
elif not invoke('real-sqlite-seed',[str(SCENARIOS/'seed.mjs'),str(candidate),str(database),str(metadata)],candidate,360):
    raise SystemExit('FAIL required seed facility unavailable; preserve evidence, do not infer product failure')
selection=os.environ.get('QA65_CASE_SELECTION','all')
if selection not in ['all','eis']:raise SystemExit('Unsupported selection')
results=[]
for root,copy in [(candidate,'candidate'),(baseline,'baseline')]:
    for flavor in ['src','dist']:
        cases=[(target,'behavior') for target in ['ae','eis','registry']] if copy=='baseline' else \
              [(target,mode) for target in ['ae','eis','registry'] for mode in ['lifecycle','retry']]+[('ae','snapshot'),('ae','explicit'),('cross','cross')]
        for target,mode in cases:
            if selection=='eis' and target!='eis':continue
            name=f'{copy}-{flavor}-{target}-{mode}'
            semantic=evidence/(name+'.json')
            ok=invoke(name,[str(SCENARIOS/'worker.mjs'),str(root),flavor,target,mode,str(semantic),str(database),str(metadata)],root,360)
            results.append(dict(name=name,result='Pass' if ok else 'Fail',semantic=semantic.name))
comparisons=[]
for flavor in ['src','dist']:
    for target in ['ae','eis','registry']:
        if selection=='eis' and target!='eis':continue
        left=evidence/f'candidate-{flavor}-{target}-lifecycle.json'
        right=evidence/f'baseline-{flavor}-{target}-behavior.json'
        if left.exists() and right.exists():
            a=json.loads(left.read_text());b=json.loads(right.read_text())
            equal=a.get('businessText')==b.get('businessText') and a['status']==b['status']=='PASS'
            comparisons.append(dict(flavor=flavor,target=target,equal=equal))
        else:comparisons.append(dict(flavor=flavor,target=target,equal=False,reason='missing semantic output'))
(evidence/'summary.json').write_text(json.dumps(dict(configuration=configuration,cases=results,comparisons=comparisons),indent=2)+'\n')
ok=all(item['result']=='Pass' for item in results) and all(item['equal'] for item in comparisons)
print('PASS' if ok else 'FAIL','independent default-owner scenarios',flush=True)
sys.exit(0 if ok else 1)
