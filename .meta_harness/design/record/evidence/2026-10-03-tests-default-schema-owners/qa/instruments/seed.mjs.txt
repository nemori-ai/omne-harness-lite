// Criterion 1/3/6 facility: independent real SQLite setup process, outside observation.
import assert from 'node:assert/strict'
import { join } from 'node:path'
import { pathToFileURL } from 'node:url'
import { writeFileSync } from 'node:fs'
const [root,database,metadata]=process.argv.slice(2)
assert.equal(process.env.QA65_EXECUTION_ALLOWED,'frozen-lane-transferred')
const support=await import(pathToFileURL(join(root,'packages/assets/evaluation-input-store/tests/support.ts')).href)
const host=await support.boot(database)
try {
  const identity=await host.identity('qa-default-owner')
  const session='session_qa_defaults'
  await host.initialize(identity.tenant,session)
  const result=await host.reader.read(session,identity.tenant)
  assert.ok('input' in result,'real seed is readable before observation')
  writeFileSync(metadata,JSON.stringify({tenant:identity.tenant,session},null,2)+'\n')
  console.log('PASS real SQLite seed')
} catch(error) {console.error('FAIL real SQLite seed',error);process.exitCode=1}
finally {await host.close(false)}
